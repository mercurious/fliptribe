cmake_minimum_required(VERSION 3.16)
project(fliptribe CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -ffast-math")
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
pkg_check_modules(SDL2_TTF REQUIRED SDL2_ttf)
file(GLOB SRC ${CMAKE_SOURCE_DIR}/src/*.cpp)
list(FILTER SRC EXCLUDE REGEX "main_synctest\\.cpp$")
add_executable(fliptribe ${SRC})
target_include_directories(fliptribe PRIVATE ${SDL2_INCLUDE_DIRS} ${SDL2_TTF_INCLUDE_DIRS})
target_link_libraries(fliptribe ${SDL2_LIBRARIES} ${SDL2_TTF_LIBRARIES} m pthread)
target_compile_options(fliptribe PRIVATE -Wall -Wextra -Wno-unused-parameter)

# host-side Wi-Fi sync interop check, run next to HI5's or Jaffedrone's synctest (no SDL; not built by default)
add_executable(fliptribe_synctest EXCLUDE_FROM_ALL src/engine.cpp src/sync.cpp src/main_synctest.cpp)
target_link_libraries(fliptribe_synctest m pthread)
target_compile_options(fliptribe_synctest PRIVATE -Wall -Wextra -Wno-unused-parameter)
